package SiCe.GUI;
import javax.swing.*;

import SiCe.GUI.authentication.LoginGUI;
import SiCe.GUI.mainPanel.MainPanelGUI;
import SiCe.GUI.authentication.AccountGUI;

import java.awt.*;


public class MainWindowGUI extends JFrame {

    AccountGUI okey =new AccountGUI(this);

    private JPanel contenedor;
    private CardLayout layout;

    //Constructor de la ventana principal
    public MainWindowGUI () {
        setSize(800,600);
        setMinimumSize(new Dimension(800,600));
        setTitle("SiCe");
        setDefaultCloseOperation(JFrame.EXIT_ON_CLOSE);
        setLocationRelativeTo(null);

        layout = new CardLayout();
        contenedor = new JPanel(layout);

        //se crean los JPanel correspondientes "llamando" (new) a un constructor (el √∫nico) que tiene de param√©tro un objeto de la clase que 
        //que construye, y requiere ingresar esta misma instancia (this), para que muestra estos mismos paneles agregados 
        //a este objeto (ventana principal).
        contenedor.add(new LoginGUI(this), "LOGIN");
        contenedor.add(new MainPanelGUI(this), "PANEL");
        contenedor.add(new AccountGUI(this),"ACCOUNT");
        
        add(contenedor);

        layout.show(contenedor, "LOGIN");
    }

    //es el m√©todo que se utiliza en los ActionListener de los botones para mostrar el respectivo JPanel.
    public void changePanel (String n){
        layout.show(contenedor, n);
    }

        //El m√©todo main que inicia el programa creando un objeto de la clase de MainWindowGUI, sigo sin entender el 
        //SwingUtilities.invokeLater
        public static void main(String[] args) {
        SwingUtilities.invokeLater(() -> {
            MainWindowGUI ventana = new MainWindowGUI();
            ventana.setVisible(true);
        });
    }
}

package SiCe.GUI.mainPanel;

import javax.swing.*;
import java.awt.*;

import SiCe.GUI.MainWindowGUI;
import SiCe.GUI.dietScanner.DietScannerGUI;
import SiCe.GUI.authentication.AccountGUI;

public class MainPanelGUI extends JPanel {

    private final JPanel sidebar;
    private final JPanel contentPanel;
    private final CardLayout cardLayout;
    JPanel topBar;


    public MainPanelGUI(MainWindowGUI n){
        setLayout(new BorderLayout());
        
        topBar = createTopBar();
        sidebar = createSidebar();

        cardLayout = new CardLayout();
        contentPanel = new JPanel(cardLayout);

        contentPanel.add(new HomeGUI(), "HOME");
        contentPanel.add(new DietScannerGUI(), "SCANNER");
        contentPanel.add(new AccountGUI(n), "ACCOUNT");

        JPanel mainArea = new JPanel(new BorderLayout());
        mainArea.add(sidebar, BorderLayout.WEST);
        mainArea.add(contentPanel, BorderLayout.CENTER);

        add(topBar, BorderLayout.NORTH);
        add(mainArea, BorderLayout.CENTER);
    }

    //M√©todos auxiliares para darle el cuerpo al constructor del MainPanelGUI

    public JPanel createSidebar(){ 
        JPanel panel = new JPanel();
        panel.setLayout(new BoxLayout(panel, BoxLayout.Y_AXIS));

        panel.setBackground(new Color(200, 235, 235));
        panel.setPreferredSize(new Dimension(190, 0));
        panel.setBorder(
           BorderFactory.createCompoundBorder(
         BorderFactory.createMatteBorder(0, 0, 0, 2, Color.GRAY),
        BorderFactory.createEmptyBorder(10, 8, 10, 8)
        ));

        JButton homeButton = new JButton("Inicio");
        JButton dietButton = new JButton("Lector de dieta");
        JButton account = new JButton("Perfil");

        // Mantener los botones alineados y al ancho del men√∫
        homeButton.setAlignmentX(LEFT_ALIGNMENT);
        dietButton.setAlignmentX(LEFT_ALIGNMENT);
        account.setAlignmentX(LEFT_ALIGNMENT);

        homeButton.setMaximumSize(
            new Dimension(Integer.MAX_VALUE, 40)
        );
        dietButton.setMaximumSize(
            new Dimension(Integer.MAX_VALUE, 40)
        );
        account.setMaximumSize(
            new Dimension(Integer.MAX_VALUE, 40)
        );

        homeButton.addActionListener(e ->
            cardLayout.show(contentPanel, "HOME")
        );

        dietButton.addActionListener(e ->
            cardLayout.show(contentPanel, "SCANNER")
        );
        account.addActionListener(e ->
            cardLayout.show(contentPanel, "ACCOUNT")
        );

        panel.add(homeButton);
        panel.add(Box.createVerticalStrut(8));
        panel.add(dietButton);
        panel.add(Box.createVerticalGlue());
        panel.add(account);

        return panel;
    }

    public JPanel createTopBar () {
        JPanel topBar = new JPanel(new FlowLayout(FlowLayout.LEFT));
        JButton menuButton = new JButton("‚ò∞");
        JLabel title = new JLabel("SiCe");

        menuButton.addActionListener(e -> {
            sidebar.setVisible(!sidebar.isVisible());

            revalidate();
            repaint();
        });

        topBar.add(menuButton);
        topBar.add(title);

        return topBar;
    }

}
